```python
async def _node_execute_tools(state: WorkflowState) -> WorkflowState:
    """图的一等工具节点。复用 framework/tools/dispatcher.py 的双路调度
    （RuleToolRouter 零延迟 + LLM function calling），不重造。

    与 shop_action_agent.py:135 的区别：那里是 Agent 内部行为、图不可见；
    这里是图节点，自动获得 _traced() 观测、可被 check_completion 回环、
    失败可路由到 recover。
    """
    from app.framework.tools.dispatcher import ToolDispatcher
    from app.providers.tools import get_tool_registry

    calls = state.pending_tool_calls or []
    if not calls:
        return state

    dispatcher = ToolDispatcher(get_tool_registry())
    for call in calls:
        res = await dispatcher.dispatch(call["name"], _tool_ctx(state), args=call.get("args"))
        state.tool_results.append({"name": call["name"], "ok": res.success, "content": res.content})
        # 工具调用轨迹进 trace_steps，前端「深度思考轨迹」直接可渲染
        state.trace_steps.append({
            "step_id": f"T{len(state.trace_steps) + 1:03d}",
            "agent_name": "execute_tools", "action": call["name"],
            "output_summary": str(res.content)[:80],
            "status": "ok" if res.success else "failed",
        })
    state.pending_tool_calls = []
    return state
```

```python
_MODE_BUDGET = {
    OrchestrationMode.STANDARD: AGENT_LOOP_MAX_ROUNDS,
    OrchestrationMode.DEEP: AGENT_LOOP_DEEP_ROUNDS,
}

def _node_check_completion(state: WorkflowState) -> WorkflowState:
    """收敛判定。DEEP 档看 todo 完成度，STANDARD 看轮次与是否重复调同一工具。"""
    budget = _MODE_BUDGET[state.mode]
    state.round_no += 1
    if state.round_no >= budget:
        state.plan["completion_next"] = "decision"       # 到上限强制收尾
    elif state.mode is OrchestrationMode.DEEP and state.todo:
        state.plan["completion_next"] = "loop" if any(not t["done"] for t in state.todo) else "decision"
    else:
        state.plan["completion_next"] = "decision" if state.answer_draft else "loop"
    return state
```

### 2.1 发现机制：从中心清单到尾缀扫描

**现状瓶颈**：`providers/tools/__init__.py:13 builtin(repo)` 里硬编码 import 21 个工具类；recall / memory / agents 各有一份同模式 `builtin()`，共 4 处中心清单。加工具 = 写类 + **改中心文件** → 多人并行时是合并冲突热点，违反开闭原则。

**已有的一半基建**：`framework/registry.py` 已有 `@component` 装饰器（挂 kind/name/priority 元数据）+ `ComponentRegistry`。缺的只是「发现」。

按 amap 的取舍实现 `os.walk` 尾缀扫描（**不用 `pkgutil.walk_packages`**——它会递归 import 每个子包触发所有 `__init__.py` 副作用）：

```python
# backend/app/framework/discovery.py（新建，约 60 行）
_SUFFIXES = ("_tool.py", "_tools.py", "_skill.py", "_source.py", "_provider.py")
_SKIP_DIRS = {"__pycache__", ".git", "node_modules", ".venv", "tests"}

def discover(package_roots: list[str]) -> int:
    """扫描文件系统、只 import 命中尾缀约定的模块，import 副作用触发 @component 注册。

    为什么不用 pkgutil.walk_packages：它递归 __import__ 每个子包来拿 __path__，
    会触发所有 __init__.py 副作用；os.walk 读目录不触发 import，只有命中尾缀的
    模块才 import_module，兄弟包完全不被碰。冷启动预算 < 50ms（与 amap 同口径）。

    错误处理分三档（对齐 amap，三档对应三种错误语义）：
      单模块 import 失败 → warning + skip（一个坏插件不能拖死启动）
      discover 后再加 root → RuntimeError（静默漏扫是最难排的故障）
      重复 discover      → no-op 幂等
    """
    if _state["done"]:
        return 0
    count = 0
    for root in package_roots:
        base = Path(__file__).resolve().parents[1] / root.replace(".", "/")
        for dirpath, dirnames, filenames in os.walk(base):
            dirnames[:] = [d for d in dirnames if d not in _SKIP_DIRS]
            for fn in filenames:
                if not fn.endswith(_SUFFIXES):
                    continue
                mod = _module_name(dirpath, fn)
                try:
                    importlib.import_module(mod)
                    count += 1
                except Exception:
                    logger.warning("discover: skip %s", mod, exc_info=True)
    _state["done"] = True
    logger.info("discover: imported %d modules", count)
    return count
```

**`builtin()` 不删**，降级为「测试注入与显式覆盖」通道——同名后注册者覆盖，测试可以塞替身。这保住了可测性，也让存量装配路径不用一次性迁完。

```python
# backend/app/framework/skills/directory.py（新建）
class DirectorySkillProvider:
    """多 root 目录聚合，同名 last-wins（目录排后者覆盖前者）——
    这是"定向覆盖"的实现方式：先挂 common/ 再挂业务目录即可定制。

    渐进披露：list_manifests() 只返回 name/description（轻量常驻 prompt），
    get_skill(name) 才读 SKILL.md 全文，references/ 由 skill_read_resource 按需读。
    """

    def __init__(self, roots: list[Path]) -> None:
        self._roots = roots
        self._index: dict[str, Path] = {}

    def refresh(self) -> None:
        for root in self._roots:                     # 顺序即优先级
            for sk in sorted(root.glob("*/SKILL.md")):
                meta, _ = split_front_matter(sk.read_text(encoding="utf-8"))
                if meta.get("enabled", True):
                    self._index[meta["name"]] = sk   # last-wins

    def list_manifests(self) -> list[dict]:
        """轻量索引：只有 name/description，控制 prompt 占用。"""
        out = []
        for name, path in self._index.items():
            meta, _ = split_front_matter(path.read_text(encoding="utf-8"))
            out.append({"name": name, "description": meta.get("description", "")})
        return out

    def get_skill(self, name: str) -> tuple[dict, str]:
        """按需加载全文 + front-matter（含 allowed-tools 白名单）。"""
        meta, body = split_front_matter(self._index[name].read_text(encoding="utf-8"))
        return meta, body
```

**allowed-tools 绑定**：skill 被选中时，`execute_tools` 节点只允许该 skill 声明的工具。这既是安全边界（穿搭 skill 不该能调 `order.pay`），也降低 LLM 选错工具的概率。

```python
# _node_execute_tools 内追加
allowed = state.active_skill_allowed_tools     # 由 plan 节点选中 skill 时写入
if allowed and call["name"] not in allowed:
    state.tool_results.append({"name": call["name"], "ok": False,
                               "content": f"当前技能未授权该工具（allowed: {sorted(allowed)}）"})
    continue
```

`framework/tools/registry.py` 现在同名工具静默覆盖。自动发现后这变得危险（两个开发者各写一个 `search_products`）：

```python
class ToolConflictPolicy(str, Enum):
    ERROR = "error"        # 默认：启动期拒绝。同名几乎总是接入失误，
```

### 2.4 验收标准：接入成本可量化

这是「可扩展性」的硬指标，写进文档供其他开发者对照：

| 场景 | 目标接入成本 |
|---|---|
| 加一个 tool | **1 个文件**：`providers/tools/xxx_tool.py`（继承基类 + `@component`），无需改任何中心清单 |
| 加一个 skill | **1 个目录**：`skills/<name>/SKILL.md`（+ 可选 `references/`），零 Python 代码 |
| 加一个召回源 | **1 个文件**：`providers/recall/xxx_source.py` |

**验证**：写一个最小的 `echo_tool.py` 与一个最小的 `demo_skill/SKILL.md`，只放文件不改别处，重启后断言两者都出现在注册表与 manifest 里。这个用例要固化成单测，防止发现机制被后续改动破坏。

## 阶段三｜卡片与文字配合

阶段一完成后，`_order_by_cited` / `_slim_evidence` 这套机制自动作用于所有路径（因为只有一条路径了）。剩下要补的是**契约化**——现在卡片与文字的绑定靠 `answer_cited_pids` 这个约定字段，没有强校验。

```python
# backend/app/schemas/workflow.py 追加
class AnswerBinding(BaseModel):
    """回答与卡片的绑定契约。让"文字驱动卡片"从约定变成校验点。"""
    cited_pids: list[str]              # 回答正文引用的商品，决定卡片顺序
    evidence_ids: list[str]            # 支撑这些引用的证据条目
    unresolved: list[str] = []         # 回答提到但候选里没有 → guard 必须拦
```

`ResponseGuard` 已有 `_check_cited_in_list`（`verification/response_guard.py:92`）在查这件事，把它的产出结构化进 `AnswerBinding.unresolved`，并让 SSE 只下发 `unresolved` 为空的绑定。收益是前端不用再自己猜卡片顺序，也不会出现「文字夸了 A、卡片首位是 B」这种最伤信任的表现。

**风险**：低。以现有 guard 为基础做结构化，不改判定逻辑。

## 阶段四｜context / memory

### 4.1 context 依赖 DAG

现状 `framework/context/manager.py` 全并行采集，Provider 之间无法传递结果。补 amap 的 `depends_on` + 拓扑分层：

```python
# framework/context/protocols.py 追加
class ContextProvider(Protocol):
    name: str
    priority: int
    depends_on: tuple[str, ...] = ()           # 硬依赖：上游失败则本层跳过
    optional_depends_on: tuple[str, ...] = ()  # 软依赖：上游失败仍执行
```

分层执行时每层成功的 slices 注入下层 `upstream_slices`，全局 deadline 跨层共享。**触发这项的真实场景已经出现**：皮肤检测 skill 需要「先取用户肤质画像 → 再据画像取护肤品上下文」，这就是链式依赖。

**代价要讲清楚**：分层执行的总延迟是各层最慢之和（全并行是全局最慢）。所以只给真有依赖的 Provider 声明 `depends_on`，其余保持同层并行。

### 4.2 memory 增补「用过即降权」

现有 3 路（preference / session / used_memories）。`used_memories` 已经在做去重，可以再加一档 amap 的 MMR 重排思路：同一轮对话里已注入过的记忆降权，避免反复强调同一条偏好（用户体感是"它老拿这件事说事"）。

优先级低于前三阶段——先把编排与热插拔做完，记忆的收益要等 skill 多了、上下文竞争激烈了才显现。

## 5. 分阶段与验收

| 阶段 | 内容 | 预估 | 风险 | 关键门禁 |
|---|---|---|---|---|
| 零 | 删除快速模式（后端 7 处 + 前端 3 处） | 0.5–1 天 | 低 | `grep FAST_MODE\|fastMode` 零残留；延迟基线已归档 |
| 一 | 统一编排底座（两档入图 + execute_tools 节点 + 收 OmniAgent） | 2–3 天 | **高** | 死循环闸门测试 + SSE 事件序列 diff 为空 + 深度思考路径过 guard |
| 二 | 发现机制 + Skill 资产库 + 冲突策略 | 2 天 | 中 | 「只加 1 个文件/目录就能接入」单测通过；CI 强制 governance |
| 三 | AnswerBinding 契约化 | 0.5 天 | 低 | 构造「回答提到候选外商品」用例，断言被拦 |
| 四 | context DAG + memory 降权 | 1–2 天 | 中 | 分层执行的 P95 延迟不高于全并行基线 20% |

**阶段零放最前的理由**：它独立于其他阶段、风险最低，且删完之后阶段一只需统一**两档**而不是三档，直接降低那个高风险阶段的复杂度。

**顺序不可调**：阶段二的 skill 需要阶段一的 `execute_tools` 节点承载 allowed-tools 边界；阶段三依赖阶段一收敛成单路径；阶段四的 DAG 是为阶段二的 skill 服务的。
